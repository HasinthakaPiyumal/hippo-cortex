\begin{algorithm}[H]
\caption{Example Algorithm}
\label{alg:example}
\begin{algorithmic}[1]
\Require Input data $D$, Parameter $k$
\Ensure Output result $R$
\State Initialize $R \gets \emptyset$
\For{each item $i$ in $D$}
    \If{condition is met}
        \State Process item $i$
        \State Add result to $R$
    \EndIf
\EndFor
\State \Return $R$
\end{algorithmic}
\end{algorithm}
